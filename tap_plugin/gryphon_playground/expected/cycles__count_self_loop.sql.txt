SELECT "tap_edge"."from_entity_id" AS "_g_col_0", COUNT("tap_edge"."to_entity_id") AS "_g_agg_0"
FROM "tap_edge"
INNER JOIN "tap_entity" ON ("tap_edge"."entity_id" = "tap_entity"."id")
INNER JOIN "tap_entity" T3 ON ("tap_edge"."from_entity_id" = T3."id")
INNER JOIN "tap_entity" T4 ON ("tap_edge"."to_entity_id" = T4."id")
WHERE ("tap_entity"."deleted_at" IS NULL AND T3."deleted_at" IS NULL AND T4."deleted_at" IS NULL AND "tap_edge"."edge_type" = %s)
GROUP BY 1
ORDER BY 1 ASC
